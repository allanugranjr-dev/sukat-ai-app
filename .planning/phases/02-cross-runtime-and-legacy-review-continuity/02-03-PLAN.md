    2. ALTER TABLE scans ADD COLUMN processing_progress_reported TINYINT(1) DEFAULT 0
    3. CREATE TABLE scan_processing_attempts with columns matching MariaDB: id (VARCHAR(36)), scan_id (VARCHAR(36)), attempt_number (INT), status (VARCHAR(20)), provider (VARCHAR(120)), processing_version (VARCHAR(80)), idempotency_key (VARCHAR(255)), claim_token (VARCHAR(36)), quality (TEXT), quality_issues (JSON), reconstruction (JSON), staged_measurements (JSON), staged_model_path (VARCHAR(500)), error_code (VARCHAR(80)), error_message (VARCHAR(1000)), is_promoted (TINYINT(1)), created_at (DATETIME), started_at (DATETIME), completed_at (DATETIME), promoted_at (DATETIME), updated_at (DATETIME)
    4. Add indexes: (scan_id, created_at DESC), (scan_id) WHERE status IN ('queued','validating','processing','retrying')
    5. Add foreign key: scans.processing_attempt_id → scan_processing_attempts(id) ON DELETE SET NULL

    Update xampp/api/index.php with PHP helper functions:

    1. Function findActiveScanAttempt(scanId): returns active attempt or null

    5. Function failScanAttempt(attemptId, errorCode, errorMessage): update attempt to failed
    6. Function safeErrorMessage(value): redaction of SQL, paths, URLs, tokens (see server/scanProcessingAttempt.mjs patterns)

    Ensure all functions use PDO prepared statements to prevent SQL injection.
